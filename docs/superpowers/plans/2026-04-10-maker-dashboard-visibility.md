```python
def _days_left(end_date_iso: str, now: float):
    """Return days until resolution as float, or None if unparseable."""
    if not end_date_iso:
        return None
    try:
        from datetime import datetime, timezone
        end_dt = datetime.fromisoformat(end_date_iso.replace("Z", "+00:00"))
        return (end_dt.timestamp() - now) / 86400.0
    except ValueError:
        return None
```

```javascript
function fmtVol(v) {
  if (!v || v === 0) return '$0';
  if (v >= 1e6) return '$' + (v / 1e6).toFixed(1) + 'M';
  if (v >= 1e3) return '$' + (v / 1e3).toFixed(1) + 'k';
  return '$' + Math.round(v);
}

function fmtDays(d) {
  if (d == null) return '—';
  if (d < 0) return 'Expired';
  if (d < 1 / 24) return Math.round(d * 24 * 60) + 'm';
  if (d < 1) return Math.round(d * 24) + 'h';
  return d.toFixed(1) + 'd';
}

```
